cmake_minimum_required(VERSION 3.25)
project(mirage_winfsp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(WINFSP_SDK_ROOT "" CACHE PATH "Extracted WinFsp SDK root")
set(MIRAGE_RUST_PROFILE "debug" CACHE STRING "Rust profile used for the FFI library")
if(NOT EXISTS "${WINFSP_SDK_ROOT}/inc/winfsp/winfsp.h")
  message(FATAL_ERROR "Set WINFSP_SDK_ROOT to the pinned WinFsp SDK root")
endif()
add_executable(mirage-fs src/main.cpp src/service.cpp src/volume.cpp src/ffi_bridge.cpp src/file_context.cpp src/directory_buffer.cpp src/status_map.cpp src/callbacks/get_volume_info.cpp src/callbacks/get_security.cpp src/callbacks/get_file_info.cpp src/callbacks/open.cpp src/callbacks/read_directory.cpp src/callbacks/mutation.cpp src/callbacks/read.cpp src/pending_read.cpp src/cancel.cpp src/security.cpp src/host_control.cpp)
if(MIRAGE_RUST_PROFILE STREQUAL "release")
  set(MIRAGE_CARGO_PROFILE_ARGS --release --locked)
else()
  set(MIRAGE_CARGO_PROFILE_ARGS)
endif()
set(MIRAGE_RUST_LIB "${CMAKE_CURRENT_SOURCE_DIR}/../../target/${MIRAGE_RUST_PROFILE}/mirage_ffi.lib")
add_custom_target(mirage-rust-ffi
  COMMAND cargo build -p mirage-ffi ${MIRAGE_CARGO_PROFILE_ARGS}
  BYPRODUCTS "${MIRAGE_RUST_LIB}"
  WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../.."
  VERBATIM)
add_dependencies(mirage-fs mirage-rust-ffi)
target_include_directories(mirage-fs PRIVATE include ../../crates/mirage-ffi/include)
target_include_directories(mirage-fs SYSTEM PRIVATE "${WINFSP_SDK_ROOT}/inc")
target_link_libraries(mirage-fs PRIVATE "${WINFSP_SDK_ROOT}/lib/winfsp-x64.lib" "${MIRAGE_RUST_LIB}" advapi32 bcrypt ntdll userenv ws2_32)
target_compile_definitions(mirage-fs PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(mirage-fs PRIVATE /W4 /WX /permissive- /EHsc /wd4324 /wd4815 /Zc:strictStrings-)
target_compile_options(mirage-fs PRIVATE "$<$<CONFIG:Release>:/Zi>" "$<$<CONFIG:Release>:/Brepro>")
target_link_options(mirage-fs PRIVATE "$<$<CONFIG:Release>:/DEBUG:FULL>" "$<$<CONFIG:Release>:/Brepro>" "$<$<CONFIG:Release>:/INCREMENTAL:NO>" "$<$<CONFIG:Release>:/OPT:REF>" "$<$<CONFIG:Release>:/OPT:ICF>" "$<$<CONFIG:Release>:/PDBALTPATH:%_PDB%>")
target_compile_definitions(mirage-fs PRIVATE "PNTSTATUS=NTSTATUS*")
